'=====================================================================
' Process  : AP_RemittancePosting
' Owner    : Finance AP automation
' Note     : pasted from iBot designer - no version info available
'=====================================================================

Option Explicit

Dim oApp, oCred, oBook, aRows, i
Dim BatchId, PolicyRef, Amount

Sub Main()
    On Error Goto Handler

    BatchId = GetQueueParameter("BatchId")
    Log INFO, "Starting remittance posting for batch " & BatchId

    Set oApp  = Robot.LaunchApplication("C:\Apps\PolCore\polcore.exe", 60000)
    Set oCred = Robot.GetCredential("SVC_POLCORE_BOT")

    Robot.WaitForWindow "<wnd app='polcore.exe' title='PolCore - Sign In' />", 30000
    Robot.SetText "<wnd app='polcore.exe' ctrlname='txtUser' />", oCred.User
    Robot.SetText "<wnd app='polcore.exe' ctrlname='txtPass' />", oCred.Password
    Robot.Click   "<wnd app='polcore.exe' ctrlname='btnLogin' />"
    Robot.WaitForWindow "<wnd app='polcore.exe' title='PolCore - Main' />", 45000

    Set oBook = Excel.Open("D:\ibot\input\batch_" & BatchId & "\remittance_" & BatchId & ".xlsx")
    aRows = Excel.ReadRange(oBook, "Sheet1!A2:H5000")

    For i = 0 To UBound(aRows)
        PolicyRef = aRows(i, 0)
        Amount    = aRows(i, 4)

        Queue.SetStatus "InProgress"

        Robot.Click   "<wnd app='polcore.exe' ctrlname='tabPolicy' idx='12' />", 15000
        Robot.SetText "<wnd app='polcore.exe' ctrlname='txtPolicyRef' />", PolicyRef
        Robot.Click   "<wnd app='polcore.exe' ctrlname='btnSearch' />", 15000

        ' grid can be slow when the mainframe bridge is busy
        Robot.WaitForElement "<wnd app='polcore.exe' ctrlname='gridResults' />", 20000

        Robot.SetText "<wnd app='polcore.exe' ctrlname='txtAmount' />", Amount
        Robot.Click   "<wnd app='polcore.exe' ctrlname='btnPost' />", 15000
        Robot.WaitForElement "<wnd app='polcore.exe' title='Posting complete' />", 30000

        Queue.SetStatus "Success"
        Log INFO, "Posted " & PolicyRef & " amount " & Amount
    Next

    Excel.Close oBook
    Robot.CloseApplication "polcore.exe"
    Exit Sub

Handler:
    Log ERROR, "Failed at row " & i & ": " & Err.Description
    Robot.CaptureScreenshot
    Queue.SetStatus "Failed"
    Err.Raise Err.Number
End Sub
