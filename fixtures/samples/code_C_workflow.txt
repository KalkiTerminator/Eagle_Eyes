<Activity x:Class="AP_RemittancePosting" xmlns="http://schemas.ibot/workflow">
  <Variables>
    <Variable x:TypeArguments="x:String" Name="BatchId" />
    <Variable x:TypeArguments="x:String" Name="PolicyRef" />
    <Variable x:TypeArguments="x:Decimal" Name="Amount" />
  </Variables>

  <Sequence DisplayName="Main">
    <LaunchApplication DisplayName="OpenPolCore"
                       Path="C:\Apps\PolCore\polcore.exe" TimeoutMS="60000" />
    <GetCredential DisplayName="FetchServiceCredential" Asset="SVC_POLCORE_BOT" />

    <RetryScope DisplayName="LoginWithRetry" NumberOfRetries="3" RetryIntervalMS="5000">
      <Sequence>
        <TypeInto Selector="&lt;wnd app='polcore.exe' ctrlname='txtUser' /&gt;" Text="[Cred.User]" />
        <TypeInto Selector="&lt;wnd app='polcore.exe' ctrlname='txtPass' /&gt;" Text="[Cred.Password]" />
        <Click    Selector="&lt;wnd app='polcore.exe' ctrlname='btnLogin' /&gt;" />
      </Sequence>
    </RetryScope>

    <ExcelApplicationScope WorkbookPath="[&quot;D:\ibot\input\batch_&quot; + BatchId]">
      <ReadRange SheetName="Sheet1" Range="A2:H5000" Output="[Rows]" />
    </ExcelApplicationScope>

    <ForEachRow DataTable="[Rows]" DisplayName="PostEachRemittance">
      <Sequence>
        <Click DisplayName="ClickPolicyTab"
               Selector="&lt;wnd app='polcore.exe' ctrlname='tabPolicy' idx='12' /&gt;" TimeoutMS="15000" />
        <TypeInto Selector="&lt;wnd app='polcore.exe' ctrlname='txtPolicyRef' /&gt;" Text="[row(0)]" />
        <Click    Selector="&lt;wnd app='polcore.exe' ctrlname='btnSearch' /&gt;" TimeoutMS="15000" />
        <WaitForElement DisplayName="WaitForResultsGrid"
                        Selector="&lt;wnd app='polcore.exe' ctrlname='gridResults' /&gt;" TimeoutMS="20000" />
        <TypeInto Selector="&lt;wnd app='polcore.exe' ctrlname='txtAmount' /&gt;" Text="[row(4).ToString]" />
        <Click    Selector="&lt;wnd app='polcore.exe' ctrlname='btnPost' /&gt;" TimeoutMS="15000" />
        <WaitForElement DisplayName="WaitForPostingConfirmation"
                        Selector="&lt;wnd app='polcore.exe' title='Posting complete' /&gt;" TimeoutMS="30000" />
      </Sequence>
    </ForEachRow>

    <TryCatch DisplayName="GlobalHandler">
      <Catches>
        <Catch x:TypeArguments="s:Exception">
          <Sequence>
            <TakeScreenshot FilePath="[ScreenshotPath]" />
            <LogMessage Level="Error" Message="[exception.ToString]" />
            <Rethrow />
          </Sequence>
        </Catch>
      </Catches>
    </TryCatch>
  </Sequence>
</Activity>
